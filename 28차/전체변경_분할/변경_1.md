+            return Err(format!("색상은 #RRGGBB 형식이어야 합니다: {s}"));

+        Ok(Self(u8::from_str_radix(&s[1..3], 16).unwrap(),

+    fn hex(self) -> String { format!("#{:02X}{:02X}{:02X}", self.0, self.1, self.2) }
